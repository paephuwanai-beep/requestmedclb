ขั้นตอนการสร้าง Google Apps Script Backend:

  1. สร้าง Google Sheet ใหม่ ตั้งชื่อว่า RequestMed_DB แล้วคัดลอกลิงก์มาใส่ช่อง "Google Sheet URL" ด้านบน
  2. ไปที่เมนู Extensions > Apps Script แล้ววางโค้ดสคริปต์ด้านล่างนี้
  3. กด Deploy > New deployment > เลือก Web app (Execute as: Me, Who has access: Anyone)
  4. คัดลอก Web App URL มาใส่ในช่อง "Web App URL" ด้านบน

Google Apps Script Code สำหรับ Google Sheets 

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  
  // If sheet is empty, add header
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["ID", "Date", "Recipient", "Name", "IDCard", "Department", "Purpose", "Status", "Timestamp"]);
  }
  
  sheet.appendRow([
    data.id,
    data.date,
    data.recipient,
    data.name,
    data.idCard,
    data.department,
    data.purposeText,
    data.status,
    new Date()
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({"status": "success"})).setMimeType(ContentService.MimeType.JSON);
}
